#define	SMU_PCTL_NAND		0xC8010208		/* SMU */

/* NAND pin function */
#define SMU_PCTL_NAND_8BIT	0x00000088
#define SMU_PULLCTL_NAND_8BIT	0x0000007f
#define SMU_PCTL_NAND_16BIT	0x000000f7
#define SMU_PULLCTL_NAND_16BIT	0x02000000

/* timeout */
#define NAND_TMO_100US  2500 /* 250MHz x 10clk¤È²¾Äê */
#define NAND_READ_TIMEOUT  (NAND_TMO_100US * 5)
#define NAND_WRITE_TIMEOUT  (NAND_TMO_100US * 100)
#define NAND_ERASE_TIMEOUT  (NAND_TMO_100US * 500)
#define NAND_ACCESS_EDO2WAIT  0x00222200 /* WE/RE  H 3clk/L 3clk */
#define NAND_ACCESS_EDO1WAIT  0x00111100 /* WE/RE  H 2clk/L 2clk */
#define NAND_ACCESS_EDO0WAIT  0x00000000 /* WE/RE  H 1clk/L 1clk  AHB <= 66MHz */
#define NAND_ACCESS_1WAIT	0x00130100 /* WEH 1,WEP 2, REH 2, REP 4 */
#define NAND_ACCESS_WAIT	0x01C52200  /* (CLH/ALH 0, CLS/ALS 1, REH 12, RP 5, WH 2, WP 2) + 1 */
#define NAND_ACCESS_TOSHIBA	0x00351100  /* (CLH/ALH 0, CLS/ALS 0, REH  3, RP 5, WH 1, WP 1) + 1 */
#define NAND_ACCESS_SAMSUNG	0x01C31200  /* (CLH/ALH 0, CLS/ALS 1, REH 12, RP 3, WH 1, WP 2) + 1 */
#define NAND_ACCESS_HYNIX	0x01C32200  /* (CLH/ALH 0, CLS/ALS 1, REH 12, RP 3, WH 2, WP 2) + 1 */

#define NAND_MSR_INIT		(NAND_MSR_FBDC|NAND_MSR_PSIZE_2048|NAND_ACCESS_WAIT)	/* 2048B/page  8bit width */
#define NAND_MSR_TOSHIBA	(NAND_MSR_FBDC|NAND_MSR_PSIZE_2048|NAND_ACCESS_TOSHIBA)	/* 2048B/page  8bit width */
#define NAND_MSR_SAMSUNG	(NAND_MSR_FBDC|NAND_MSR_PSIZE_2048|NAND_ACCESS_SAMSUNG) /* 2048B/page  8bit width */
#define NAND_MSR_HYNIX		(NAND_MSR_FBDC|NAND_MSR_PSIZE_2048|NAND_ACCESS_HYNIX)   /* 2048B/page  8bit width */

/* ID */
